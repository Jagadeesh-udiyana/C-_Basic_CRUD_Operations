CREATE Procedure USP_SEL_ALL_Employees            
AS            
Begin            -- exec USP_SEL_ALL_Employees       
          
 SELECT EM.EMPGUID, EM.EMPName, EM.DepartmentID, EM.MobileNo , EM.EmailID, EM.JoinDate, EM.Salary, EM.Status           
 FROM Employee_Master EM WITH (NOLOCK)              
            
 SELECT EE.EDU_GUID, EE.EMPGUID, EE.Qualification, EE.InstitutionName, EE.YearOfPassing, EE.Percentage, EE.Remarks           
 FROM Employee_Education EE WITH (NOLOCK)             
             
End
